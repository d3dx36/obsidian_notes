class MainActivity : AppCompatActivity() {  
    override fun onCreate(savedInstanceState: Bundle?) {  
      super.onCreate(savedInstanceState)  
        setContentView(R.layout.activity_main)  Указываем путь к  экрану
        val content=findViewById<ViewGroup>(R.id.content)  ищем контейнер
        repeat(100){  
            val newItem=layoutInflater.inflate(R.layout.item, content)  путь к создаваемому элементу, куда разместить элемент
        }  
    }  
}

В итоге имеем такую общую формулу:

**`inflate(<R.layout.X>, <Контейнер>, <true / false>)`** , где **X** — наименование файла разметки.

Используя такую схему можно создать и добавить в контейнер (`container1`) XML файл разметки `item.xml`:

val container1 = findViewById<ViewGroup>(R.id.container1)
val view = layoutInflater.inflate(R.layout.item, container1, false //false - не отображает элементы хотя они созданы)
container1.addView(view)
